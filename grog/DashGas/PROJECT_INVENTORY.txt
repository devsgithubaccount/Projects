USAGE QUICK START
================================================================================

Database Usage:
  from database import DashGasDatabase
  db = DashGasDatabase("dashgas.db")
  supplier_id = db.add_supplier("Shell", "shell@example.com", "555-0100", 2)
  product_id = db.add_product("Regular Gas", "fuel", 10000, "gallons", 2000, 
                              supplier_id, 2, unit_cost=2.50)
  db.log_stock_change(product_id, 9500, -500, "consumption")
  summary = db.get_inventory_summary()
  db.close()

Simulator Usage:
  from simulators import GasStationInventorySimulator
  station = GasStationInventorySimulator()
  station.simulate_days(7)
  report = station.get_inventory_report()
  json_output = station.get_all_products_json()
  alerts = station.get_low_stock_alerts(threshold_percent=25)

Integration Example:
  db = DashGasDatabase("dashgas.db")
  station = GasStationInventorySimulator()
  # Simulate consumption and log to database
  for day in range(30):
      results = station.simulate_day()
      for product_id, simulator in station.simulators.items():
          product = db.get_product_by_name(simulator.product_name)
          if product:
              db.log_stock_change(product['product_id'], 
                                  simulator.current_stock, 
                                  simulator.consumption_history[-1] * -1,
                                  "daily_consumption")
